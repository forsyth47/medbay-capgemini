  Future<void> _loadData() async {
    final uid = SupabaseService.currentUserId!;
    final m = await SupabaseService.getMedicines(uid);
    final a = await SupabaseService.getAlerts(forCurrentUser: true);

      medicines = m;
      schedules = s;
      alerts = a.take(2).toList();
      device = d;
      loading = false;
    });
    _startCountdown();
  }

  TimeOfDay? _parseTime(String t) {
    try {
      final parts = t.split(' ');
      if (parts.length != 2) return null;
      final hm = parts[0].split(':');
      if (hm.length != 2) return null;
      var hour = int.parse(hm[0]);
      final minute = int.parse(hm[1]);
      if (parts[1] == 'PM' && hour != 12) hour += 12;
      if (parts[1] == 'AM' && hour == 12) hour = 0;
      return TimeOfDay(hour: hour, minute: minute);
    } catch (_) {
      return null;
    }
  }

  DateTime _timeToDateTime(String timeStr) {
    final tod = _parseTime(timeStr);
    if (tod == null) return DateTime.now();
    final now = DateTime.now();
    return DateTime(now.year, now.month, now.day, tod.hour, tod.minute);
  }

  Schedule? get _nextSchedule {
    final now = DateTime.now();
    final sorted = [...schedules]
      ..sort(
        (a, b) => _timeToDateTime(a.time).compareTo(_timeToDateTime(b.time)),
      );
    for (final sch in sorted) {
      if (_timeToDateTime(sch.time).isAfter(now)) return sch;
    }
    return sorted.isNotEmpty ? sorted.first : null;
  }

  int get _takenDoses {
    final now = DateTime.now();
    return schedules.where((s) => _timeToDateTime(s.time).isBefore(now)).length;
  }

  int get _totalDoses => schedules.length;

  void _startCountdown() {
    _timer?.cancel();
    _updateCountdown();
    _timer = Timer.periodic(
      const Duration(seconds: 1),
      (_) => _updateCountdown(),
    );
  }

  void _updateCountdown() {
    final next = _nextSchedule;
    if (next == null) return;
    var target = _timeToDateTime(next.time);
    final now = DateTime.now();
    if (!target.isAfter(now)) {
      target = target.add(const Duration(days: 1));
    }
    if (mounted) setState(() => _timeRemaining = target.difference(now));
  }

  String get _countdownText {
    final h = _timeRemaining.inHours.toString().padLeft(2, '0');
    final m = (_timeRemaining.inMinutes % 60).toString().padLeft(2, '0');
    final s = (_timeRemaining.inSeconds % 60).toString().padLeft(2, '0');
    return '$h:$m:$s';
  }

  String _formatSyncTime(DateTime? lastSeen) {
    if (lastSeen == null) return 'Never';
    final diff = DateTime.now().difference(lastSeen);
    if (diff.inMinutes < 1) return 'Just now';
    if (diff.inMinutes < 60) return '${diff.inMinutes}m ago';
    if (diff.inHours < 24) return '${diff.inHours}h ago';
    return '${diff.inDays}d ago';
  }

  Color _syncColor(DateTime? lastSeen) {
    if (lastSeen == null) return Colors.red;
    final diff = DateTime.now().difference(lastSeen);
    if (diff.inMinutes < 5) return Colors.green;
    if (diff.inHours < 1) return Colors.orange;
    return Colors.red;
  }

    );

      margin: const EdgeInsets.only(bottom: 8),
